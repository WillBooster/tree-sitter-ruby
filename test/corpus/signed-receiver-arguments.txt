========================================
Unary plus arguments on explicit receivers
========================================

obj.foo +x
obj&.foo +x, y
Obj::Foo +x
obj.foo +1

---

(program
  (call (identifier) (identifier) (argument_list (unary (identifier))))
  (call (identifier) (identifier) (argument_list (unary (identifier)) (identifier)))
  (call (constant) (constant) (argument_list (unary (identifier))))
  (call (identifier) (identifier) (argument_list (unary (integer)))))
